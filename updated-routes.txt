import os
import json
import sys

import yaml
import logging
from typing import Dict, Any, List, Optional, Tuple, Union

logger = logging.getLogger(__name__)


class OpenAPIParser:
    """
    Parser for OpenAPI specification files that extracts structured information
    for semantic search and retrieval
    """

    def __init__(self):
        self.supported_extensions = ['.json', '.yaml', '.yml']

    def parse_spec(self, file_path: str) -> Dict[str, Any]:
        """
        Parse an OpenAPI specification file into a dictionary

        Args:
            file_path: Path to the OpenAPI specification file

        Returns:
            Dictionary representation of the OpenAPI specification
        """
        logger.info(f"Parsing OpenAPI spec: {file_path}")

        extension = os.path.splitext(file_path)[1].lower()
        if extension not in self.supported_extensions:
            raise ValueError(
                f"Unsupported file extension: {extension}. Supported extensions: {self.supported_extensions}")

        try:
            with open(file_path, 'r', encoding='utf-8') as file:
                if extension == '.json':
                    spec = json.load(file)
                else:  # YAML files
                    spec = yaml.safe_load(file)

            # Validate that it's an OpenAPI spec
            if 'openapi' not in spec and 'swagger' not in spec:
                raise ValueError("File does not appear to be a valid OpenAPI specification")

            return spec

        except Exception as e:
            logger.error(f"Error parsing OpenAPI specification: {str(e)}")
            raise

    def extract_endpoints(self, spec: Dict[str, Any]) -> List[Dict[str, Any]]:
        """
        Extract API endpoints from an OpenAPI specification

        Args:
            spec: Dictionary representation of the OpenAPI specification

        Returns:
            List of endpoint dictionaries
        """
        endpoints = []

        try:
            if 'paths' not in spec:
                logger.warning("No paths found in OpenAPI specification")
                return endpoints

            for path, path_item in spec['paths'].items():
                # Skip parameters at path level
                if path == 'parameters':
                    continue

                # Extract path-level parameters
                path_parameters = path_item.get('parameters', [])

                # Process each HTTP method
                for method, operation in path_item.items():
                    # Skip non-HTTP methods
                    if method in ['parameters', '$ref', 'summary', 'description', 'servers']:
                        continue

                    # Skip non-dictionary operations (e.g., might be a $ref)
                    if not isinstance(operation, dict):
                        continue

                    # Build endpoint data
                    endpoint = {
                        'path': path,
                        'method': method.upper(),
                        'operation_id': operation.get('operationId', f"{method}_{path}"),
                        'summary': operation.get('summary', ''),
                        'description': operation.get('description', ''),
                        'tags': operation.get('tags', []),
                        'deprecated': operation.get('deprecated', False)
                    }

                    # Combine path and operation parameters
                    all_parameters = path_parameters.copy()
                    if 'parameters' in operation:
                        all_parameters.extend(operation['parameters'])

                    # Resolve any parameter references
                    resolved_parameters = []
                    for param in all_parameters:
                        if '$ref' in param:
                            resolved_param = self._resolve_reference(param['$ref'], spec)
                            if resolved_param:
                                resolved_parameters.append(resolved_param)
                        else:
                            resolved_parameters.append(param)

                    endpoint['parameters'] = resolved_parameters

                    # Add request body if present
                    if 'requestBody' in operation:
                        endpoint['requestBody'] = self._process_request_body(operation['requestBody'], spec)

                    # Process responses
                    if 'responses' in operation:
                        endpoint['responses'] = self._process_responses(operation['responses'], spec)

                    # Add to endpoints list
                    endpoints.append(endpoint)

            return endpoints

        except Exception as e:
            logger.error(f"Error extracting endpoints: {str(e)}")
            raise

    def _process_request_body(self, request_body: Dict[str, Any], spec: Dict[str, Any]) -> Dict[str, Any]:
        """Process request body information"""
        if '$ref' in request_body:
            request_body = self._resolve_reference(request_body['$ref'], spec)

        processed_body = {
            'description': request_body.get('description', ''),
            'required': request_body.get('required', False),
            'content': {}
        }

        if 'content' in request_body:
            for media_type, media_info in request_body['content'].items():
                processed_body['content'][media_type] = {
                    'schema': self._process_schema(media_info.get('schema', {}), spec)
                }

                if 'example' in media_info:
                    processed_body['content'][media_type]['example'] = media_info['example']

                if 'examples' in media_info:
                    processed_body['content'][media_type]['examples'] = media_info['examples']

        return processed_body

    def _process_responses(self, responses: Dict[str, Any], spec: Dict[str, Any]) -> Dict[str, Any]:
        """Process response information"""
        processed_responses = {}

        for status_code, response in responses.items():
            if '$ref' in response:
                response = self._resolve_reference(response['$ref'], spec)

            processed_response = {
                'description': response.get('description', ''),
                'content': {}
            }

            if 'content' in response:
                for media_type, media_info in response['content'].items():
                    processed_response['content'][media_type] = {
                        'schema': self._process_schema(media_info.get('schema', {}), spec)
                    }

                    if 'example' in media_info:
                        processed_response['content'][media_type]['example'] = media_info['example']

                    if 'examples' in media_info:
                        processed_response['content'][media_type]['examples'] = media_info['examples']

            processed_responses[status_code] = processed_response

        return processed_responses

    def _process_schema(self, schema: Dict[str, Any], spec: Dict[str, Any]) -> Dict[str, Any]:
        """Process schema information, resolving references"""
        if not schema:
            return {}

        if '$ref' in schema:
            resolved_schema = self._resolve_reference(schema['$ref'], spec)
            if not resolved_schema:
                return {'$ref': schema['$ref']}

            # To avoid circular references, return only the reference
            # and basic information from the resolved schema
            return {
                '$ref': schema['$ref'],
                'type': resolved_schema.get('type', 'object'),
                'description': resolved_schema.get('description', '')
            }

        # For arrays, process the items
        if schema.get('type') == 'array' and 'items' in schema:
            schema['items'] = self._process_schema(schema['items'], spec)

        # For objects, process properties
        if schema.get('type') == 'object' and 'properties' in schema:
            processed_properties = {}
            for prop_name, prop_schema in schema['properties'].items():
                processed_properties[prop_name] = self._process_schema(prop_schema, spec)
            schema['properties'] = processed_properties

        return schema

    def _resolve_reference(self, ref: str, spec: Dict[str, Any]) -> Optional[Dict[str, Any]]:
        """
        Resolve a JSON Reference within the specification

        Args:
            ref: Reference string (e.g., "#/components/schemas/Pet")
            spec: OpenAPI specification dictionary

        Returns:
            Resolved reference object or None if not found
        """
        if not ref.startswith('#/'):
            logger.warning(f"External references not supported: {ref}")
            return None

        path_parts = ref[2:].split('/')
        current = spec

        try:
            for part in path_parts:
                current = current[part]
            return current
        except (KeyError, TypeError):
            logger.warning(f"Could not resolve reference: {ref}")
            return None

    def prepare_for_vector_db(self, spec: Dict[str, Any]) -> List[Dict[str, Any]]:
        """
        Prepare OpenAPI specification for storage in a vector database

        Args:
            spec: OpenAPI specification dictionary

        Returns:
            List of documents ready for vector database
        """
        vector_docs = []

        # Add general API information
        if 'info' in spec:
            info = spec['info']
            info_text = f"""
            API: {info.get('title', 'Untitled API')}
            Version: {info.get('version', 'Unknown')}
            Description: {info.get('description', 'No description available')}

            This document contains general information about the API.
            """

            vector_docs.append({
                "text": info_text,
                "metadata": {
                    "document_type": "openapi_info",
                    "content_type": "api_overview",
                    "api_title": info.get('title', 'Untitled API'),
                    "api_version": info.get('version', 'Unknown')
                }
            })

        # Process endpoints
        endpoints = self.extract_endpoints(spec)
        for endpoint in endpoints:
            # Create endpoint document
            endpoint_text = f"""
            API Endpoint: {endpoint['method']} {endpoint['path']}

            Summary: {endpoint.get('summary', 'No summary available')}

            Description: {endpoint.get('description', 'No description available')}

            Parameters: {self._format_parameters_for_text(endpoint.get('parameters', []))}
            """

            # Convert tags to string for metadata
            tags = endpoint.get("tags", [])
            if tags and isinstance(tags, list):
                tags = ",".join(tags)

            vector_docs.append({
                "text": endpoint_text,
                "metadata": {
                    "document_type": "openapi_endpoint",
                    "content_type": "endpoint_description",
                    "method": endpoint["method"],
                    "path": endpoint["path"],
                    "tags": tags,
                    "deprecated": endpoint.get("deprecated", False)
                }
            })

            # Create request body document if present
            if "requestBody" in endpoint and endpoint["requestBody"]:
                request_body_text = self._format_request_body_for_text(endpoint["requestBody"])

                vector_docs.append({
                    "text": f"""
                    API Endpoint Request Body: {endpoint['method']} {endpoint['path']}

                    {request_body_text}
                    """,
                    "metadata": {
                        "document_type": "openapi_endpoint_request",
                        "content_type": "request_body",
                        "method": endpoint["method"],
                        "path": endpoint["path"]
                    }
                })

            # Create response documents
            if "responses" in endpoint and endpoint["responses"]:
                for status_code, response_info in endpoint["responses"].items():
                    response_text = self._format_response_for_text(status_code, response_info)

                    vector_docs.append({
                        "text": f"""
                        API Endpoint Response ({status_code}): {endpoint['method']} {endpoint['path']}

                        {response_text}
                        """,
                        "metadata": {
                            "document_type": "openapi_endpoint_response",
                            "content_type": "response",
                            "method": endpoint["method"],
                            "path": endpoint["path"],
                            "status_code": status_code
                        }
                    })

        # Process components/schemas if present
        if "components" in spec and "schemas" in spec["components"]:
            for schema_name, schema in spec["components"]["schemas"].items():
                schema_text = self._format_schema_for_text(schema_name, schema)

                vector_docs.append({
                    "text": f"""
                    API Schema: {schema_name}

                    {schema_text}
                    """,
                    "metadata": {
                        "document_type": "openapi_schema",
                        "content_type": "schema",
                        "schema_name": schema_name
                    }
                })

        return vector_docs

    def _format_parameters_for_text(self, parameters: List[Dict[str, Any]]) -> str:
        """Format parameters for text representation"""
        if not parameters:
            return "No parameters required."

        param_texts = []
        for param in parameters:
            required = "Required" if param.get("required", False) else "Optional"
            schema_type = "unknown"

            if "schema" in param:
                schema = param["schema"]
                if "type" in schema:
                    schema_type = schema["type"]
                elif "$ref" in schema:
                    schema_type = f"Reference: {schema['$ref']}"

            param_text = f"- {param.get('name', 'Unnamed')} ({param.get('in', 'unknown')}, {schema_type}, {required}): {param.get('description', 'No description')}"
            param_texts.append(param_text)

        return "\n".join(param_texts)

    def _format_request_body_for_text(self, request_body: Dict[str, Any]) -> str:
        """Format request body for text representation"""
        description = request_body.get("description", "No description available")
        required = "Required" if request_body.get("required", False) else "Optional"

        content_texts = []
        for media_type, media_info in request_body.get("content", {}).items():
            content_texts.append(f"Content Type: {media_type}")

            schema = media_info.get("schema", {})
            if schema:
                content_texts.append(self._format_schema_properties(schema))

            if "example" in media_info:
                try:
                    example_str = json.dumps(media_info["example"], indent=2)
                    content_texts.append(f"Example:\n{example_str}")
                except:
                    content_texts.append("Example: Complex example available")

        return f"{description} ({required})\n\n" + "\n".join(content_texts)

    def _format_response_for_text(self, status_code: str, response_info: Dict[str, Any]) -> str:
        """Format response for text representation"""
        description = response_info.get("description", "No description available")

        content_texts = []
        for media_type, media_info in response_info.get("content", {}).items():
            content_texts.append(f"Content Type: {media_type}")

            schema = media_info.get("schema", {})
            if schema:
                content_texts.append(self._format_schema_properties(schema))

            if "example" in media_info:
                try:
                    example_str = json.dumps(media_info["example"], indent=2)
                    content_texts.append(f"Example:\n{example_str}")
                except:
                    content_texts.append("Example: Complex example available")

        return f"Status Code {status_code}: {description}\n\n" + "\n".join(content_texts)

    def _format_schema_for_text(self, schema_name: str, schema: Dict[str, Any]) -> str:
        """Format schema for text representation"""
        texts = []

        description = schema.get("description", "No description available")
        schema_type = schema.get("type", "object")

        texts.append(f"Type: {schema_type}")
        texts.append(f"Description: {description}")

        if schema_type == "object" and "properties" in schema:
            texts.append("\nProperties:")
            for prop_name, prop_info in schema["properties"].items():
                prop_type = prop_info.get("type", "unknown")
                prop_desc = prop_info.get("description", "No description")
                required = "Required" if prop_name in schema.get("required", []) else "Optional"

                texts.append(f"- {prop_name} ({prop_type}, {required}): {prop_desc}")

        elif schema_type == "array" and "items" in schema:
            items = schema["items"]
            items_type = items.get("type", "unknown")
            if "$ref" in items:
                items_type = f"Reference: {items['$ref']}"

            texts.append(f"\nArray Items Type: {items_type}")

        return "\n".join(texts)

    def _format_schema_properties(self, schema: Dict[str, Any]) -> str:
        """Format schema properties for text representation"""
        texts = []

        if "$ref" in schema:
            return f"Schema: Reference to {schema['$ref']}"

        schema_type = schema.get("type", "unknown")
        texts.append(f"Schema Type: {schema_type}")

        if schema_type == "object" and "properties" in schema:
            texts.append("Properties:")
            for prop_name, prop_info in schema["properties"].items():
                prop_type = prop_info.get("type", "unknown")
                if "$ref" in prop_info:
                    prop_type = f"Reference: {prop_info['$ref']}"

                prop_desc = prop_info.get("description", "No description")
                required = "Required" if prop_name in schema.get("required", []) else "Optional"

                texts.append(f"- {prop_name} ({prop_type}, {required}): {prop_desc}")

        elif schema_type == "array" and "items" in schema:
            items = schema["items"]
            items_type = items.get("type", "unknown")
            if "$ref" in items:
                items_type = f"Reference: {items['$ref']}"

            texts.append(f"Array Items Type: {items_type}")

        return "\n".join(texts)
