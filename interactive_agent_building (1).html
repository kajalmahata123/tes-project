from langchain.agents import create_openai_tools_agent, AgentExecutor
from langchain_openai import ChatOpenAI

llm = ChatOpenAI(model="gpt-4")

# 4. Create executor
executor = AgentExecutor(agent=agent, tools=tools)
# 5. Run!
result = executor.invoke({"input": "What's the weather in Tokyo?"})

from langchain.memory import ConversationBufferWindowMemory from langchain.schema import BaseMessage class MemoryAgent: def __init__(self): # Remember last 10 interactions self.memory = ConversationBufferWindowMemory( k=10, return_messages=True ) # Long-term preferences storage self.user_preferences = {} # Learning from interactions self.success_patterns = [] self.failure_patterns = [] async def chat(self, user_input: str, user_id: str): """Chat with memory and learning""" # Get conversation history history = self.memory.chat_memory.messages # Get user preferences preferences = self.user_preferences.get(user_id, {}) # Create context-aware prompt full_context = f""" User preferences: {preferences} Recent conversation: {history[-3:] if history else "None"} New message: {user_input} """ # Process with full context response = await self.agent.ainvoke({"input": full_context}) # Save to memory self.memory.save_context( {"input": user_input}, {"output": response["output"]} ) # Learn from interaction await self._learn_from_interaction(user_input, response["output"]) return response["output"] async def _learn_from_interaction(self, input_msg, output_msg): """Learn patterns from successful interactions""" # Simple learning: track what works if any(word in output_msg.lower() for word in ["thank", "great", "perfect"]): self.success_patterns.append({ "input_pattern": input_msg.lower(), "successful_response": output_msg })

from typing import TypedDict

class HomeState(TypedDict):
user_request: str # What user wants
time_of_day: str # Morning, evening, etc.
weather: str # Hot, cold, etc.
actions_taken: list # What we already did
# Step 3: Create thinking functions
def analyze_situation(state):
"""Look at the situation and understand what's needed"""
# Check what time it is, what weather is like, etc.
return state
def make_smart_decision(state):
"""Decide what to do based on the situation"""
if state["time_of_day"] == "evening" and state["weather"] == "cold":
return "cozy_mode" # Warm and dim
elif state["time_of_day"] == "morning":
return "energize_mode" # Bright and fresh
else:
return "comfort_mode" # Just comfortable
# Step 4: Build the smart workflow
workflow = StateGraph(HomeState)
# Add the thinking steps
workflow.add_node("analyze", analyze_situation)
workflow.add_node("decide", make_smart_decision)
workflow.add_node("cozy_mode", make_cozy)

smart_home_agent = workflow.compile()
